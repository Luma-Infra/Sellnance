$branch = "  " + [char]0x251C + [char]0x2500 + [char]0x2500 + " "
$lastBranch = "  " + [char]0x2514 + [char]0x2500 + [char]0x2500 + " "

$diffList = [System.Collections.Generic.List[string]]::new()
$tableList = [System.Collections.Generic.List[string]]::new()

$diffList.Add("# Codebase Lines & Function Metrics")
$diffList.Add("")
$diffList.Add("> VS Code Color Guide: File (+ Green), Function (! Cyan/Blue)")
$diffList.Add("")
$diffList.Add('```diff')

$tableList.Add("## Table View (Markdown Preview)")
$tableList.Add("")
$tableList.Add("| File Path / Function | Line Count | Char Count |")
$tableList.Add("| :--- | ---: | ---: |")

$allFiles = Get-ChildItem -Recurse -File | Where-Object {
    $_.Name -notlike "@*" -and (
        (($_.FullName -like "*\modules\*") -and ($_.Extension -eq ".py" -or $_.Extension -eq ".go")) -or 
        (($_.FullName -like "*\static\*") -and ($_.Extension -eq ".js" -or $_.Extension -eq ".css")) -or 
        (($_.FullName -like "*\templates\*") -and ($_.Extension -eq ".html")) -or 
        (($_.DirectoryName -eq $pwd.Path) -and ($_.Extension -in ".py", ".js", ".json", ".go", ".txt"))
    )
}

foreach ($fileItem in $allFiles) {
    $filePathRel = $fileItem.FullName.Replace((Get-Location).Path + "\", "")
    $ext = $fileItem.Extension
    $isMin = $fileItem.Name -like "*.min.*"
    
    $content = Get-Content $fileItem.FullName -Encoding utf8
    $cleanLines = $content | Where-Object {
        $trimmed = $_.Trim()
        $trimmed -ne "" -and 
        -not $trimmed.StartsWith("#") -and 
        -not $trimmed.StartsWith("//") -and
        -not $trimmed.StartsWith("/*") -and
        -not $trimmed.StartsWith("*") -and
        -not $trimmed.EndsWith("*/") -and
        -not ($trimmed.StartsWith('"""') -or $trimmed.StartsWith("'''"))
    }
    
    $charCount = 0
    foreach ($line in $cleanLines) {
        $charCount += $line.Replace(" ", "").Replace("`t", "").Length
    }

    # 1. Diff block row (+: Green)
    $paddedPath = $filePathRel.PadRight(52)
    $linesStr = ($cleanLines.Count.ToString()).PadLeft(5)
    $charsStr = ($charCount.ToString()).PadLeft(7)
    $diffList.Add("+ $paddedPath [$linesStr lines | $charsStr chars]")

    # 2. Markdown Table row
    $tableList.Add("| **``" + $filePathRel + "``** | **" + $cleanLines.Count + "** | **" + ("{0:N0}" -f $charCount) + "** |")

    # Extract functions
    if (-not $isMin -and ($ext -in ".py", ".js", ".go")) {
        $funcs = @()
        $lines = $content

        # Python
        if ($ext -eq ".py") {
            for ($i = 0; $i -lt $lines.Count; $i++) {
                if ($lines[$i] -match '^(\s*)(?:async\s+def|def)\s+([a-zA-Z0-9_]+)') {
                    $indent = $matches[1].Length
                    $name = $matches[2] + "()"
                    $start = $i
                    $bodyStart = $i
                    while ($bodyStart -lt $lines.Count -and -not ($lines[$bodyStart].Trim().EndsWith(":"))) {
                        $bodyStart++
                    }
                    $bodyStart++
                    $end = $bodyStart
                    for ($j = $bodyStart; $j -lt $lines.Count; $j++) {
                        $next = $lines[$j]
                        $t = $next.Trim()
                        if ($t -ne "" -and -not $t.StartsWith("#")) {
                            $m = [regex]::Match($next, '^(\s*)')
                            if ($m.Groups[1].Value.Length -le $indent) { break }
                            $end = $j
                        }
                    }
                    $funcs += [PSCustomObject]@{ Name = $name; Start = $start; End = $end }
                }
            }
        } 
        # JS & Go
        elseif ($ext -in ".js", ".go") {
            for ($i = 0; $i -lt $lines.Count; $i++) {
                $line = $lines[$i]
                $name = ""
                if ($ext -eq ".go" -and $line -match '^\s*func\s+(?:\([^)]+\)\s+)?([a-zA-Z0-9_]+)') {
                    $name = $matches[1] + "()"
                } elseif ($ext -eq ".js") {
                    if ($line -match '^\s*(?:export\s+)?(?:default\s+)?(?:async\s+)?function\s*([a-zA-Z0-9_]+)') {
                        $name = $matches[1] + "()"
                    } elseif ($line -match '^\s*(?:export\s+)?(?:const|let|var)\s+([a-zA-Z0-9_]+)\s*=\s*(?:async\s*)?(?:\([^)]*\)|[a-zA-Z0-9_]+)?\s*=>') {
                        $name = $matches[1] + "()"
                    } elseif ($line -match '^\s*(?:export\s+)?(?:const|let|var)\s+([a-zA-Z0-9_]+)\s*=\s*(?:async\s*)?function') {
                        $name = $matches[1] + "()"
                    } elseif ($line -match '^\s*(?:window\.)?([a-zA-Z0-9_]+)\s*=\s*(?:async\s*)?function') {
                        $name = $matches[1] + "()"
                    }
                }
                if ($name) {
                    $start = $i
                    $braceCount = 0
                    $foundBrace = $false
                    $end = $i
                    for ($j = $i; $j -lt $lines.Count; $j++) {
                        $l = $lines[$j]
                        $openB = ($l -replace '[^{]', '').Length
                        $closeB = ($l -replace '[^}]', '').Length
                        if ($openB -gt 0) { $foundBrace = $true }
                        $braceCount += $openB - $closeB
                        if ($foundBrace -and $braceCount -le 0) {
                            $end = $j
                            break
                        }
                    }
                    $funcs += [PSCustomObject]@{ Name = $name; Start = $start; End = $end }
                }
            }
        }

        # Add sub-functions
        for ($k = 0; $k -lt $funcs.Count; $k++) {
            $fn = $funcs[$k]
            $isLast = ($k -eq ($funcs.Count - 1))
            $branchSymbol = if ($isLast) { $lastBranch } else { $branch }
            $charBranch = if ($isLast) { [char]0x2514 } else { [char]0x251C }
            $tablePrefix = "&nbsp;&nbsp;" + $charBranch + [char]0x2500 + [char]0x2500 + " "

            $fnLines = $lines[$fn.Start .. $fn.End]
            $fnClean = $fnLines | Where-Object {
                $trimmed = $_.Trim()
                $trimmed -ne "" -and 
                -not $trimmed.StartsWith("#") -and 
                -not $trimmed.StartsWith("//") -and
                -not $trimmed.StartsWith("/*") -and
                -not $trimmed.StartsWith("*") -and
                -not $trimmed.EndsWith("*/") -and
                -not ($trimmed.StartsWith('"""') -or $trimmed.StartsWith("'''"))
            }
            $fnChars = 0
            foreach ($l in $fnClean) {
                $fnChars += $l.Replace(" ", "").Replace("`t", "").Length
            }

            # 1. Diff block row (!: Cyan/Blue)
            $paddedFn = ($branchSymbol + $fn.Name).PadRight(52)
            $fnLinesStr = ($fnClean.Count.ToString()).PadLeft(5)
            $fnCharsStr = ($fnChars.ToString()).PadLeft(7)
            $diffList.Add("! $paddedFn [$fnLinesStr lines | $fnCharsStr chars]")

            # 2. Table row
            $tableList.Add("| " + $tablePrefix + "``" + $fn.Name + "`` | " + $fnClean.Count + " | " + ("{0:N0}" -f $fnChars) + " |")
        }
    }
}

$diffList.Add('```')
$diffList.Add("")

$finalOutput = [System.Collections.Generic.List[string]]::new()
$finalOutput.AddRange($diffList)
$finalOutput.AddRange($tableList)

$finalOutput | Out-File -FilePath "@lines_for_ai.md" -Encoding utf8

Write-Host "[SUCCESS] '@lines_for_ai.md' color report generated!" -ForegroundColor Green
